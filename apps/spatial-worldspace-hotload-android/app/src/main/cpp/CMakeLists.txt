cmake_minimum_required(VERSION 3.22.1)
project(spatial_gpu_workload LANGUAGES CXX)
option(RQ_SESSION_CORE "Compose the separately qualified canonical session core" OFF)
if(RQ_SESSION_CORE)
  if(NOT ANDROID_STL STREQUAL "c++_shared")
    message(FATAL_ERROR "Selected session core requires the packaged shared NDK C++ runtime")
  endif()
  if(NOT RQ_EXPERIMENT_SERVICE_INCLUDE OR NOT EXISTS "${RQ_EXPERIMENT_SERVICE_INCLUDE}/rq_experiment_services.h")
    message(FATAL_ERROR "Selected session core requires its exact canonical ABI header")
  endif()
  if(NOT RQ_SESSION_COMPOSITION_BINDING MATCHES "^[a-f0-9]+$")
    message(FATAL_ERROR "Selected session core requires the compiled activation binding")
  endif()
  string(LENGTH "${RQ_SESSION_COMPOSITION_BINDING}" RQ_SESSION_BINDING_LENGTH)
  if(NOT RQ_SESSION_BINDING_LENGTH EQUAL 64)
    message(FATAL_ERROR "Selected session activation binding must contain exactly 64 lowercase hex digits")
  endif()
  set(RQ_EXPERIMENT_ACTIVATION_BINDING_SHA256 "${RQ_SESSION_COMPOSITION_BINDING}")
endif()
set_property(GLOBAL PROPERTY JOB_POOLS rq_workload_serial=1)
set(CMAKE_JOB_POOL_COMPILE rq_workload_serial)
set(CMAKE_JOB_POOL_LINK rq_workload_serial)
option(RQ_XR_SUBMISSION_OBSERVER "Select debug-only submission observer and reader" OFF)
if(RQ_XR_SUBMISSION_OBSERVER)
  if(NOT RQ_XR_SUBMISSION_OBSERVER_MODULE OR NOT EXISTS "${RQ_XR_SUBMISSION_OBSERVER_MODULE}/submission_observer_status.h")
    message(FATAL_ERROR "Selected observer requires an explicitly staged module")
  endif()
  if(NOT RQ_OPENXR_INCLUDE_DIR OR NOT EXISTS "${RQ_OPENXR_INCLUDE_DIR}/openxr/openxr_loader_negotiation.h")
    message(FATAL_ERROR "Selected observer requires explicit official negotiation headers")
  endif()
  add_subdirectory("${RQ_XR_SUBMISSION_OBSERVER_MODULE}" "${CMAKE_CURRENT_BINARY_DIR}/submission-observer")
  add_library(rq_xr_submission_observer_probe SHARED "${CMAKE_CURRENT_SOURCE_DIR}/../../debug/cpp/submission_observer_probe.cpp")
  target_include_directories(rq_xr_submission_observer_probe PRIVATE "${RQ_XR_SUBMISSION_OBSERVER_MODULE}")
  target_compile_features(rq_xr_submission_observer_probe PRIVATE cxx_std_20)
  target_compile_options(rq_xr_submission_observer_probe PRIVATE -Wall -Wextra -Werror)
  set_target_properties(rq_xr_submission_observer_probe PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  target_link_options(rq_xr_submission_observer_probe PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
  target_link_libraries(rq_xr_submission_observer_probe PRIVATE dl)
endif()
find_program(GLSLC glslc HINTS "${ANDROID_NDK}/shader-tools/windows-x86_64" "${ANDROID_NDK}/shader-tools/linux-x86_64" "${ANDROID_NDK}/shader-tools/darwin-x86_64" REQUIRED)
foreach(STAGE vert frag)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/atlas.${STAGE}.inc"
    COMMAND "${GLSLC}" --target-env=vulkan1.1 -Werror -mfmt=c "${CMAKE_CURRENT_SOURCE_DIR}/atlas.${STAGE}" -o "${CMAKE_CURRENT_BINARY_DIR}/atlas.${STAGE}.inc"
    DEPENDS "atlas.${STAGE}" JOB_POOL rq_workload_serial VERBATIM)
endforeach()
if(NOT RQ_WORKLOAD_HAND_SHADER OR NOT EXISTS "${RQ_WORKLOAD_HAND_SHADER}")
  message(FATAL_ERROR "Explicit public hand skin shader is required")
endif()
if(NOT RQ_WORKLOAD_OPENXR_INCLUDE OR NOT EXISTS "${RQ_WORKLOAD_OPENXR_INCLUDE}/openxr/openxr.h")
  message(FATAL_ERROR "Explicit OpenXR header source is required")
endif()
foreach(SHADER gpu_hand_skinning gpu_hand_pack)
  if(SHADER STREQUAL "gpu_hand_skinning")
    set(INPUT_SHADER "${RQ_WORKLOAD_HAND_SHADER}")
  else()
    set(INPUT_SHADER "${CMAKE_CURRENT_SOURCE_DIR}/gpu_hand_pack.comp")
  endif()
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${SHADER}.comp.inc"
    COMMAND "${GLSLC}" --target-env=vulkan1.1 -fshader-stage=compute -Werror -mfmt=c "${INPUT_SHADER}" -o "${CMAKE_CURRENT_BINARY_DIR}/${SHADER}.comp.inc"
    DEPENDS "${INPUT_SHADER}" JOB_POOL rq_workload_serial VERBATIM)
endforeach()
set(RQ_WORKLOAD_PUBLIC_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}")
option(RQ_EXTERNAL_WORKLOAD_HOTLOAD "Enable copied external workload kernels in explicit debug builds" OFF)
if(RQ_EXTERNAL_WORKLOAD_HOTLOAD AND NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
  message(FATAL_ERROR "External workload hotload is debug-only")
endif()
if(RQ_WORKLOAD_EXTERNAL_SOURCE_DIR)
  if(NOT EXISTS "${RQ_WORKLOAD_EXTERNAL_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Selected external workload source is absent")
  endif()
  add_subdirectory("${RQ_WORKLOAD_EXTERNAL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/external-workload")
else()
  add_library(rq_external_workload STATIC unavailable_workload.cpp)
  target_compile_features(rq_external_workload PRIVATE cxx_std_17)
endif()
option(RQ_HOTLOAD_TESTS "Enable fixed diagnostic hotload fixtures in this debug test app" OFF)
if(RQ_HOTLOAD_TESTS)
  if(NOT RQ_HOTLOAD_CATALOG_INCLUDE OR NOT EXISTS "${RQ_HOTLOAD_CATALOG_INCLUDE}/hotload_catalog.h")
    message(FATAL_ERROR "Precompiled fixed kernel catalog is required for the debug hotload fixture")
  endif()
  set(HOTLOAD_SOURCE hotload_kernel.cpp)
else()
  set(HOTLOAD_SOURCE hotload_disabled.cpp)
endif()
add_library(spatial_gpu_workload SHARED atlas.cpp gpu_hand_input.cpp gpu_hand_workload.cpp ${HOTLOAD_SOURCE}
  "${CMAKE_CURRENT_BINARY_DIR}/atlas.vert.inc" "${CMAKE_CURRENT_BINARY_DIR}/atlas.frag.inc"
  "${CMAKE_CURRENT_BINARY_DIR}/gpu_hand_skinning.comp.inc" "${CMAKE_CURRENT_BINARY_DIR}/gpu_hand_pack.comp.inc")
target_include_directories(spatial_gpu_workload PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${RQ_WORKLOAD_OPENXR_INCLUDE}" "${RQ_HOTLOAD_CATALOG_INCLUDE}")
target_compile_features(spatial_gpu_workload PRIVATE cxx_std_17)
target_compile_definitions(spatial_gpu_workload PRIVATE VK_USE_PLATFORM_ANDROID_KHR)
target_compile_options(spatial_gpu_workload PRIVATE -Wall -Wextra -Werror)
target_link_options(spatial_gpu_workload PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(spatial_gpu_workload PRIVATE rq_external_workload android log vulkan)

if(RQ_EXTERNAL_WORKLOAD_HOTLOAD)
  target_compile_definitions(rq_external_workload PUBLIC RQ_EXTERNAL_WORKLOAD_HOTLOAD=1)
endif()
